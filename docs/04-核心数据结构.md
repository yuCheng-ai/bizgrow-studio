# 04. 核心数据结构

本章节定义 BizGrow Studio 的核心数据结构，覆盖本体、业务对象、关系、状态机、规则、Agent、业务闭环模板与运行实例。

---

### 1. OntologyConcept
```ts
type OntologyConcept = {
  id: string;
  name: string;
  code: string;
  description: string;
  attributes: string[];
  actions: string[];
  relations: string[];
};
```

### 2. BusinessObjectType
```ts
type BusinessObjectType = {
  id: string;
  conceptId: string;
  name: string;
  code: string;
  fields: FieldDef[];
  actions: ActionDef[];
  stateMachineId?: string;
  agentPolicyIds: string[];
};
```

### 3. BusinessObjectInstance
```ts
type BusinessObjectInstance = {
  id: string;
  type: string;
  data: Record<string, unknown>;
  state?: string;
  createdAt: string;
  updatedAt: string;
};
```

### 4. SemanticRelation
```ts
type SemanticRelation = {
  id: string;
  fromType: string;
  relation: string;
  toType: string;
  meaning: string;
  condition?: string;
};
```

### 5. StateMachine
```ts
type StateMachine = {
  id: string;
  objectType: string;
  states: string[];
  transitions: StateTransition[];
};

type StateTransition = {
  from: string;
  to: string;
  action: string;
  guard?: string[];
  requireHumanConfirm?: boolean;
};
```

### 6. Rule
```ts
type Rule = {
  id: string;
  name: string;
  when: string;
  conditions: string[];
  actions: RuleAction[];
  priority: number;
  enabled: boolean;
};
```

### 7. AgentPolicy
```ts
type AgentPolicy = {
  id: string;
  agentId: string;
  canRead: string[];
  canExecute: string[];
  cannotExecute: string[];
  requireHumanConfirmWhen: string[];
  outputs: string[];
};
```

### 8. BusinessLoopTemplate
```ts
type BusinessLoopTemplate = {
  id: string;
  name: string;
  goal: string;
  primaryObjectType: string;
  stages: LoopStage[];
  startFrom: string;
  endWhen: string[];
};

type LoopStage = {
  id: string;
  name: string;
  type: 'human' | 'system' | 'agent' | 'condition';
  objectType?: string;
  agentId?: string;
  input?: string[];
  output?: string[];
  triggerWhen?: string[];
  requireHumanConfirm?: boolean;
};
```

### 9. BusinessLoopInstance
```ts
type BusinessLoopInstance = {
  id: string;
  templateId: string;
  name: string;
  goal: string;
  primaryObject: { type: string; id: string };
  relatedObjects: Array<{ type: string; id: string; role: string }>;
  currentStageId: string;
  status: 'running' | 'waiting_human' | 'blocked' | 'completed' | 'failed';
  context: Record<string, unknown>;
  startedAt: string;
  updatedAt: string;
  owner?: string;
  auditLogIds: string[];
};
```

### 10. Task
```ts
type Task = {
  id: string;
  loopInstanceId: string;
  title: string;
  source: 'rule' | 'agent' | 'system' | 'human';
  targetObject?: { type: string; id: string };
  status: 'pending' | 'processing' | 'done' | 'cancelled';
  ownerRole?: string;
  deadline?: string;
  requireHumanConfirm: boolean;
};
```

### 11. AuditLog
```ts
type AuditLog = {
  id: string;
  loopInstanceId: string;
  timestamp: string;
  actor: 'human' | 'agent' | 'system';
  actorId: string;
  event: string;
  reads?: string[];
  rulesHit?: string[];
  action?: string;
  before?: unknown;
  after?: unknown;
  result: 'success' | 'waiting' | 'failed';
  humanConfirmed?: boolean;
};
```
